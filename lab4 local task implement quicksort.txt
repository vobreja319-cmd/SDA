
#include <iostream>
#include <vector>
#include <utility>
using namespace std;

int partition(vector<int>& values, int lo, int hi) {
    int pivot = values[lo];
    int boundary = lo;

    for (int scan = lo + 1; scan < hi; scan++) {
        if (values[scan] < pivot) {
            boundary++;
            swap(values[boundary], values[scan]);
        }
    }

    swap(values[lo], values[boundary]);
    return boundary;
}

void quicksort(vector<int>& values, int lo, int hi) {
    if (hi - lo <= 1) {
        return;
    }

    int p = partition(values, lo, hi);

    quicksort(values, lo, p);
    quicksort(values, p + 1, hi);
}

int main() {
    vector<int> values = {6, 2, 5, 2, 1};

    quicksort(values, 0, values.size());

    for (int x : values) {
        cout << x << " ";
    }

    cout << endl;
    return 0;
}
