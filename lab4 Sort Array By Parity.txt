
#include <vector>
using namespace std;

class Solution {
public:
    vector<int> sortArrayByParity(vector<int>& nums) {
        int boundary = 0;

        for (int scan = 0; scan < nums.size(); scan++) {
            if (nums[scan] % 2 == 0) {
                swap(nums[boundary], nums[scan]);
                boundary++;
            }
        }

        return nums;
    }
};
