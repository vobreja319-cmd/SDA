
#include <vector>
using namespace std;

class Solution {
public:
    int searchInsert(vector<int>& values, int target) {
        int lo = 0;
        int hi = values.size();

        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;

            if (values[mid] < target) {
                lo = mid + 1;
            } else {
                hi = mid;
            }
        }

        return lo;
    }
};
