#include <vector>
using namespace std;

class Solution {
private:
    vector<int> buffer;

    void mergeSort(vector<int>& nums, int left, int right) {
        if (left >= right) {
            return;
        }

        int mid = left + (right - left) / 2;

        mergeSort(nums, left, mid);
        mergeSort(nums, mid + 1, right);

        int i = left;
        int j = mid + 1;
        int k = left;

        while (i <= mid && j <= right) {
            if (nums[i] <= nums[j]) {
                buffer[k++] = nums[i++];
            } else {
                buffer[k++] = nums[j++];
            }
        }

        while (i <= mid) {
            buffer[k++] = nums[i++];
        }

        while (j <= right) {
            buffer[k++] = nums[j++];
        }

        for (int p = left; p <= right; p++) {
            nums[p] = buffer[p];
        }
    }

public:
    vector<int> sortArray(vector<int>& nums) {
        if (nums.empty()) {
            return nums;
        }

        buffer.resize(nums.size());
        mergeSort(nums, 0, static_cast<int>(nums.size()) - 1);

        return nums;
    }
};
